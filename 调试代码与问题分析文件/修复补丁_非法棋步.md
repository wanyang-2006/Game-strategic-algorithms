```cpp
// 验证棋盘的整体合法性
bool Chess::validateBoardLegality() {
    // 检查1: 恰好有2个将/帅
    int redKingCount = 0, blackKingCount = 0;
    
    // 检查2: 象在己方半场，士在九宫
    for (int x = 0; x < BOARDWIDTH; x++) {
        for (int y = 0; y < BOARDHEIGHT; y++) {
            if (currTurnId < 0 || currTurnId >= state.size()) {
                return false;
            }
            
            auto &grid = state[currTurnId][xy2pos(x, y)];
            
            if (grid.color == EMPTY || grid.type == None) {
                continue;
            }
            
            // 计数将帅
            if (grid.type == King) {
                if (grid.color == RED) redKingCount++;
                else if (grid.color == BLACK) blackKingCount++;
                
                // 将帅必须在九宫内
                if (!inKingArea(x, y, grid.color)) {
                    return false;
                }
            }
            
            // 象必须在己方半场
            if (grid.type == Bishop) {
                if (!inColorArea(x, y, grid.color)) {
                    return false;
                }
            }
            
            // 士必须在九宫内
            if (grid.type == Assistant) {
                if (!inKingArea(x, y, grid.color)) {
                    return false;
                }
            }
        }
    }
    
    // 必须恰好有2个将帅
    if (redKingCount != 1 || blackKingCount != 1) {
        return false;
    }
    
    return true;
}

// 验证棋子移动的约束
bool Chess::validatePieceConstraints(const Move &move) {
    if (!inBoard(move.source_x, move.source_y) || 
        !inBoard(move.target_x, move.target_y)) {
        return false;
    }
    
    int sourcePos = xy2pos(move.source_x, move.source_y);
    int targetPos = xy2pos(move.target_x, move.target_y);
    
    auto &source = state[currTurnId][sourcePos];
    auto &target = state[currTurnId][targetPos];
    
    // 源必须是己方棋子
    if (source.color != currColor || source.type == None) {
        return false;
    }
    
    // 目标不能是己方棋子
    if (target.color == currColor) {
        return false;
    }
    
    // 根据棋子类型验证具体约束
    switch (source.type) {
        case Bishop: // 象
            // 象不能过河
            if (!inColorArea(move.target_x, move.target_y, currColor)) {
                return false;
            }
            // 象眼必须畅通
            {
                int midX = (move.source_x + move.target_x) / 2;
                int midY = (move.source_y + move.target_y) / 2;
                auto &eye = state[currTurnId][xy2pos(midX, midY)];
                if (eye.color != EMPTY) {
                    return false;
                }
            }
            break;
            
        case Knight: // 马
            // 马脚必须畅通
            {
                int fx, fy;
                int dx = move.target_x - move.source_x;
                int dy = move.target_y - move.source_y;
                
                if (abs(dx) == 2 && abs(dy) == 1) {
                    fx = move.source_x + dx/2;
                    fy = move.source_y;
                } else if (abs(dx) == 1 && abs(dy) == 2) {
                    fx = move.source_x;
                    fy = move.source_y + dy/2;
                } else {
                    return false; // 不是马的走法
                }
                
                auto &foot = state[currTurnId][xy2pos(fx, fy)];
                if (foot.color != EMPTY) {
                    return false;
                }
            }
            break;
            
        case Assistant: // 士
        case King: // 将帅
            // 必须在九宫内
            if (!inKingArea(move.target_x, move.target_y, currColor)) {
                return false;
            }
            break;
            
        case Pawn: // 兵卒
            {
                int forwardOne = (source.color == RED ? 1 : -1);
                bool hasCrossedRiver = (source.color == RED ? 
                                       (move.source_y > 4) : 
                                       (move.source_y < 5));
                
                // 未过河时只能向前
                if (!hasCrossedRiver) {
                    if (move.target_x != move.source_x || 
                        move.target_y != move.source_y + forwardOne) {
                        return false;
                    }
                }
            }
            break;
            
        default:
            break;
    }
    
    return true;
}
```

## 修复方案 3：修改 makeMoveAssumeLegal()

在执行移动前添加验证:

```cpp
bool Chess::makeMoveAssumeLegal(const Move &move) {
    // 先做最基本的防御检查
    if (!inBoard(move.source_x, move.source_y) || 
        !inBoard(move.target_x, move.target_y)) {
        return false;
    }
    
    // ✅ 添加约束验证
    if (!validatePieceConstraints(move)) {
        return false;
    }
    
    if (state.empty() || currTurnId < 0 || currTurnId >= state.size()) {
        return false;
    }
    if (continuePeaceTurn.empty() || currTurnId >= continuePeaceTurn.size()) {
        return false;
    }

    int sourcePos = xy2pos(move.source_x, move.source_y);
    int targetPos = xy2pos(move.target_x, move.target_y);
    if (sourcePos < 0 || sourcePos >= 90 || targetPos < 0 || targetPos >= 90) {
        return false;
    }
    if (state[currTurnId].size() != 90) {
        return false;
    }

    const Grid &sourceGridBefore = state[currTurnId][sourcePos];
    const Grid &targetGridBefore = state[currTurnId][targetPos];

    if (sourceGridBefore.color != currColor || sourceGridBefore.type == None) {
        return false;
    }
    if (targetGridBefore.color == currColor) {
        return false;
    }

    int nextPeaceTrun = continuePeaceTurn[currTurnId] + 1;
    vector<Grid> copyState = state[currTurnId];
    state.push_back(copyState);
    currTurnId++;

    auto &sourceGrid = state[currTurnId][xy2pos(move.source_x, move.source_y)];
    auto &targetGrid = state[currTurnId][xy2pos(move.target_x, move.target_y)];
    if (targetGrid.color == oppColor()) {
        nextPeaceTrun = 0;
        lastMoveEaten.push_back(true);
    } else {
        lastMoveEaten.push_back(false);
    }
    
    targetGrid.type = sourceGrid.type;
    targetGrid.color = sourceGrid.color;
    assert(sourceGrid.color == currColor);

    sourceGrid.color = EMPTY;
    sourceGrid.type = None;

    continuePeaceTurn.push_back(nextPeaceTrun);
    
    // ✅ 移动后验证棋盘合法性
    if (!validateBoardLegality()) {
        undoMove();
        return false;
    }
    
    currColor = oppColor();
    return true;
}
```

```cpp
void Chess::resetBoard() {

    // ... (其他初始化) ...

        currState.push_back(boardInfo[pos2x(mpos)][pos2y(mpos)]);
    }
    
    state.clear();
    state.push_back(currState);
```
